- [ ] **Step B1.4: Update the `listTypes` implementation to return `{ types, issues }`**

Modify the `listTypes` function body inside `createMakeService`. Replace the current return `ok(schemas)` pattern with:

```ts
async function listTypes(): Promise<Result<ListTypesResult, MakeError>> {
    const settings = getSettings();
    const folderExists = await ports.vault.exists(settings.typesFolder);
    if (!folderExists) return ok({ types: [], issues: [] });

    const listResult = await ports.vault.list(settings.typesFolder);
    if (listResult.kind === 'err') {
        return err({ kind: 'vault-error', cause: String(listResult.error) });
    }

    const prefix = settings.typesFolder.endsWith('/') ? settings.typesFolder : `${settings.typesFolder}/`;
    const jsonPaths = listResult.value.filter((p) =>
        p.startsWith(prefix) &&
        !p.slice(prefix.length).includes('/') &&
        p.endsWith('.json'),
    );

    const types: TypeSchema[] = [];
    const issues: CorruptTypeRef[] = [];

    await Promise.all(jsonPaths.map(async (path) => {
        const read = await ports.vault.read(path);
        if (read.kind === 'err') {
            issues.push({
                path,
                filename: basename(path),
                error: { kind: 'io-error', cause: String(read.error) },
            });
            return;
        }
        const parsed = parseTypeSchema(read.value.content);
        if (parsed.kind === 'err') {
            issues.push({
                path,
                filename: basename(path),
                error: parsed.error,
            });
            return;
        }
        types.push(parsed.value);
    }));

    return ok({ types, issues });
}
```

Add `basename` helper — in `make-service.ts` near the top:

```ts
function basename(path: string): string {
    const idx = path.lastIndexOf('/');
    return idx === -1 ? path : path.slice(idx + 1);
```

```ts
// src/infrastructure/obsidian/workspace-adapter.ts
import { TFile, type App, type PaneType } from 'obsidian';
import type { WorkspacePort, OpenFileMode } from '../../domain/shared/workspace-port.js';
import { err, ok, type Result } from '../../domain/shared/result.js';

const MODE_MAP: Record<OpenFileMode, PaneType | boolean> = {
	current: false,
	tab:     'tab',
	split:   'split',
};

export class ObsidianWorkspaceAdapter implements WorkspacePort {
	constructor(private readonly app: App) {}

	async openFile(path: string, mode: OpenFileMode): Promise<Result<void, string>> {
		const file = this.app.vault.getAbstractFileByPath(path);
		if (!(file instanceof TFile)) return err(`not-found: ${path}`);
		await this.app.workspace.getLeaf(MODE_MAP[mode]).openFile(file);
		return ok(undefined);
	}
}
```

```vue
function errorFor(fieldName: string): FieldError | null {
    return [...clientErrors.value, ...props.serverErrors].find((e) => e.fieldName === fieldName) ?? null;
}

function errorMessage(fieldName: string): string {
    const err = errorFor(fieldName);
    return err ? errorMessageFor(err) : '';

    return t(`make.form.errors.${err.kind}`, err as Record<string, unknown>);

function onSubmit() {
    // Client-side validation: per-kind validateValue + required checks
    const errors: FieldError[] = [];
    for (const field of props.schema.fields) {
        const raw = values[field.name];
        if (field.required && (raw === undefined || raw === null || raw === '')) {
            errors.push({ kind: 'required-missing', fieldName: field.name });
            continue;
        }
        if (raw === undefined || raw === null || raw === '') continue;
        const validated = FIELD_KINDS[field.kind].validateValue(field as never, raw);
        if (validated.kind === 'err') errors.push(validated.error);
    }
    if (props.schema.titleFieldName === null && explicitFilename.value.trim() === '') {
        errors.push({ kind: 'invalid-text', fieldName: '__filename__' });
    }
    clientErrors.value = errors;
    if (errors.length > 0) return;

    emit('submit', {
        raw: { ...values },
        explicitFilename: props.schema.titleFieldName === null ? explicitFilename.value : null,
    });
}
```

```ts
    const source = this.app.vault.getAbstractFileByPath(oldPath);
```

```ts
import { validateInstanceValues } from '../../domain/make/instance-ops.js';
import { renderInstanceContent } from '../../domain/make/instance-ops.js';
import { resolveInstancePath } from '../../domain/make/instance-ops.js';

async function createInstance(
    typeId: string,
    raw: Record<string, unknown>,
    explicitFilename: string | null,
    options?: CreateInstanceOptions,
): Promise<Result<InstanceRef, MakeError>> {
    const loaded = await loadType(typeId);
    if (loaded.kind === 'err') return loaded;
    const schema = loaded.value;

    const validated = validateInstanceValues(schema, raw);
    if (validated.kind === 'err') {
        return err({ kind: 'invalid-values', issues: validated.error as NonEmptyArray<FieldError> });
    }

    const resolved = resolveInstancePath(schema, validated.value, explicitFilename);
    if (resolved.kind === 'err') {
        if (resolved.error === 'no-title-field-and-no-filename') return err({ kind: 'no-title-field' });
        if (resolved.error === 'invalid-filename') return err({
            kind: 'invalid-values',
            issues: [{ kind: 'invalid-text', fieldName: '__filename__' }] as NonEmptyArray<FieldError>,
        });
        return err({ kind: 'vault-error', cause: String(resolved.error) });
    }
    const path = resolved.value;

    const exists = await ports.vault.exists(path);
    if (exists && options?.overwrite !== true) {
        return err({ kind: 'instance-exists', path });
    }

    const rendered = renderInstanceContent(schema, validated.value);

    const writeResult = exists
        ? await ports.vault.update(path, rendered.fullMarkdown)
        : await ports.vault.create(path, rendered.fullMarkdown);
    if (writeResult.kind === 'err') {
        return err({ kind: 'vault-error', cause: String(writeResult.error) });
    }

    const stat = await ports.vault.read(path);
    const createdAt = stat.kind === 'ok' ? new Date(stat.value.stat.ctime).toISOString() : new Date().toISOString();
    const updatedAt = stat.kind === 'ok' ? new Date(stat.value.stat.mtime).toISOString() : createdAt;

    const ref: InstanceRef = {
        typeId,
        path,
        title: basename(path).replace(/\.md$/, ''),
        createdAt,
        updatedAt,
    };

    ports.eventBus.emit('make:instance-created', { typeId, path });
    return ok(ref);
}
```

```ts
// src/ui/pages/make/use-create-instance-flow.ts
import { ref, type Ref } from 'vue';
import type { FieldError } from '../../../domain/make/errors.js';
import type { TypeId } from '../../../domain/make/types.js';
import type { MakeStore } from '../../stores/make-store.js';   // adjust to actual export

type Pending = { raw: Record<string, unknown>; explicitFilename: string | null; path: string };

export function useCreateInstanceFlow(typeId: Ref<TypeId>, store: MakeStore) {
    const submitting = ref(false);
    const serverErrors = ref<readonly FieldError[]>([]);
    const overwriteDialog = ref<Pending | null>(null);

    async function submit(payload: { raw: Record<string, unknown>; explicitFilename: string | null }): Promise<void> {
        submitting.value = true;
        serverErrors.value = [];
        const result = await store.createInstance(typeId.value, payload.raw, payload.explicitFilename);
        submitting.value = false;
        if (result.kind === 'ok') return;
        if (result.error.kind === 'invalid-values') {
            serverErrors.value = result.error.issues;
            return;
        }
        if (result.error.kind === 'instance-exists') {
            overwriteDialog.value = { raw: payload.raw, explicitFilename: payload.explicitFilename, path: result.error.path };
            return;
        }
        // no-title-field / vault-error / type-not-found: upstream notification
        // (caller surfaces via notifications composable)
    }

    async function confirmOverwrite(): Promise<void> {
        if (overwriteDialog.value === null) return;
        const pending = overwriteDialog.value;
        submitting.value = true;
        const result = await store.createInstance(typeId.value, pending.raw, pending.explicitFilename, { overwrite: true });
        submitting.value = false;
        overwriteDialog.value = null;
        if (result.kind === 'err' && result.error.kind === 'invalid-values') {
            serverErrors.value = result.error.issues;
        }
    }

    function cancelOverwrite() {
        overwriteDialog.value = null;
    }

    return { submitting, serverErrors, overwriteDialog, submit, confirmOverwrite, cancelOverwrite };
}
```

```ts
function dirname(path: string): string {
    const idx = path.lastIndexOf('/');

function inferTypeId(path: string, typesCache: readonly TypeSchema[]): string | null {
    const parent = dirname(path);
    const match = typesCache.find((s) => s.instancesFolder.replace(/\/$/, '') === parent);
    return match?.id ?? null;
```

```ts
async function deleteType(
    typeId: string,
    options: DeleteTypeOptions,
): Promise<Result<DeleteTypeReport, MakeError>> {
    const loaded = await loadType(typeId);
    if (loaded.kind === 'err') return loaded;
    const schema = loaded.value;

    let instancesDeleted = 0;
    const instanceFailures: FailedDelete[] = [];

    if (options.alsoDeleteInstances) {
        const list = await listInstances(typeId);
        if (list.kind === 'err') return list;
        for (const ref of list.value) {
            const del = await ports.vault.delete(ref.path);
            if (del.kind === 'err') {
                instanceFailures.push({ path: ref.path, cause: String(del.error) });
            } else {
                instancesDeleted += 1;
            }
        }
    }

    let baseFileDeleted = false;
    if (options.alsoDeleteBaseFile && schema.baseFile !== undefined) {
        const del = await ports.vault.delete(schema.baseFile.path);
        if (del.kind === 'ok') baseFileDeleted = true;
    }

    const jsonPath = `${getSettings().typesFolder}/${schema.id}.json`;
    const delJson = await ports.vault.delete(jsonPath);
    if (delJson.kind === 'err') return err({ kind: 'vault-error', cause: String(delJson.error) });

    ports.eventBus.emit('make:type-deleted', { typeId: schema.id, name: schema.name });
    return ok({ instancesDeleted, instanceFailures, baseFileDeleted });
}
```

⚠ **Gotcha from spec §4.6 / §6.2** — the pre-move enumeration MUST use `prev.instancesFolder` (the value from the loaded schema BEFORE the patch is applied), NOT `nextSchema.instancesFolder` (which already points at the new, empty folder). The in-memory patched schema would yield zero instances and incorrectly skip the move prompt.

Recommended implementation shape:

```ts
async function updateType(
    typeId: string,
    patch: TypeSchemaPatch,
    options?: UpdateTypeOptions,
): Promise<Result<UpdateTypeResult, MakeError>> {
    // 1. Existing validation path (invalid-schema / field-rename-warning / duplicate-name).
    // ... reuse from Chunk 3 ...

    // 2. Compute next schema and detect folder change.
    const prev = loadedSchema;
    const nextSchema: TypeSchema = { ...prev, ...patch, updatedAt: nowIso() };
    const prevFolder = prev.instancesFolder.trim();
    const nextFolder = nextSchema.instancesFolder.trim();
    const folderChanged = prevFolder !== nextFolder;

    if (!folderChanged) {
        // existing Chunk 3 JSON-write + emit path
        await writeTypeJson(nextSchema);
        ports.eventBus.emit('make:type-updated', { typeId, name: nextSchema.name, schema: nextSchema });
        return ok({ schema: nextSchema });
    }

    // 3. Enumerate OLD folder (NOT from nextSchema.instancesFolder).
    const oldInstances = await listInstancesInFolder(prevFolder, typeId);
    //            ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ MUST use prevFolder

    if (oldInstances.length === 0) {
        await writeTypeJson(nextSchema);
        ports.eventBus.emit('make:type-updated', { typeId, name: nextSchema.name, schema: nextSchema });
        return ok({ schema: nextSchema });
    }

    if (options?.moveInstances !== true) {
        return err({
            kind: 'instances-move-required',
            oldFolder: prevFolder,
            newFolder: nextFolder,
            count: oldInstances.length,
        });
    }

    // 4. Move loop — per-file isolation, no abort on failure.
    const failedMoves: FailedMove[] = [];
    let movedCount = 0;
    for (const ref of oldInstances) {
        const newPath = `${nextFolder}/${basename(ref.path)}`;
        const rename = await ports.vault.rename(ref.path, newPath);
        if (rename.kind === 'err') {
            failedMoves.push({ path: ref.path, cause: rename.error });   // verbatim adapter message (spec §6.5)
            continue;
        }
        movedCount += 1;
    }

    // 5. Write JSON even if some moves failed — schema truth wins.
    const writeResult = await writeTypeJson(nextSchema);
    if (writeResult.kind === 'err') return err({ kind: 'vault-error', cause: String(writeResult.error) });

    const moveReport: MoveReport = { oldFolder: prevFolder, newFolder: nextFolder, movedCount, failedMoves };

    ports.eventBus.emit('make:instances-moved', {
        typeId, oldFolder: prevFolder, newFolder: nextFolder, movedCount, failedCount: failedMoves.length,
    });
    ports.eventBus.emit('make:type-updated', { typeId, name: nextSchema.name, schema: nextSchema });

    if (failedMoves.length > 0) return err({ kind: 'partial-move', moveReport });
    return ok({ schema: nextSchema, moveReport });
}
```
